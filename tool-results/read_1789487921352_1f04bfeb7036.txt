   291→function parseCsvLine(line: string): string[] {
   292→  const out: string[] = [];
   293→  let cur = "";
   294→  let inQuotes = false;
   295→  for (let i = 0; i < line.length; i++) {
   296→    const ch = line[i] ?? "";
   297→    if (inQuotes) {
   298→      if (ch === '"') {
   299→        if (line[i + 1] === '"') {
   300→          cur += '"';
   301→          i++;
   302→        } else {
   303→          inQuotes = false;
   304→        }
   305→      } else {
   306→        cur += ch;
   307→      }
   308→    } else if (ch === '"') {
   309→      inQuotes = true;
   310→    } else if (ch === ",") {
   311→      out.push(cur.trim());
   312→      cur = "";
   313→    } else {
   314→      cur += ch;
   315→    }
   316→  }
   317→  out.push(cur.trim());
   318→  return out;
   319→}

   327→function median(values: number[]): number {
   328→  if (values.length === 0) return 0;
   329→  const sorted = [...values].sort((a, b) => a - b);
   330→  const mid = Math.floor(sorted.length / 2);
   331→  if (sorted.length % 2 === 1) {
   332→    return sorted[mid] ?? 0;
   333→  }
   334→  const a = sorted[mid - 1] ?? 0;
   335→  const b = sorted[mid] ?? 0;
   336→  return (a + b) / 2;
   337→}

   339→function assessBadges(input: {
   340→  edgesPerNode: number;
   341→  pctNoInEdge: number;
   342→  pctSingleTrack: number;
   343→  medianTrackLen: number;
   344→  divisionRate: number;
   345→  nodesPerFrame: number;
   346→}): BadgeAssessment[] {
   347→  const out: BadgeAssessment[] = [];
   348→
   349→  const e = input.edgesPerNode;
   350→  out.push({
   351→    key: "edgesPerNode",
   352→    label: "cạnh/node",
   353→    level: e >= 0.8 ? "good" : e >= 0.65 ? "warn" : "bad",
   354→    value: e.toFixed(2),
   355→  });
   356→
   357→  const p = input.pctNoInEdge;
   358→  out.push({
   359→    key: "noInEdge",
   360→    label: "%node không cạnh vào",
   361→    level: p < 20 ? "good" : p <= 35 ? "warn" : "bad",
   362→    value: `${p.toFixed(1)}%`,
   363→  });
   364→
   365→  const q = input.pctSingleTrack;
   366→  out.push({
   367→    key: "singleTrack",
   368→    label: "%track 1-node",
   369→    level: q < 8 ? "good" : q <= 15 ? "warn" : "bad",
   370→    value: `${q.toFixed(1)}%`,
   371→  });
   372→
   373→  const m = input.medianTrackLen;
   374→  out.push({
   375→    key: "medianTrack",
   376→    label: "trung vị track",
   377→    level: m >= 20 ? "good" : m >= 10 ? "warn" : "bad",
   378→    value: `${Number.isInteger(m) ? m : m.toFixed(1)} khung`,
   379→  });
   380→
   381→  const d = input.divisionRate;
   382→  const dense = input.nodesPerFrame >= 15;
   383→  let level: BadgeLevel;
   384→  let hint: string | undefined;
   385→  if (d > 60) {
   386→    level = "bad";
   387→    hint = "nhiễu merge-split";
   388→  } else if (dense && d === 0) {
   389→    level = "bad";
   390→    hint = "bỏ sót phân bào";
   391→  } else if (dense && d >= 2 && d <= 25) {
   392→    level = "good";
   393→  } else if (!dense) {
   394→    level = "warn";
   395→    hint = "mật độ node/khung < 15 — chỉ tham khảo";
   396→  } else if (d < 2) {
   397→    level = "warn";
   398→    hint = "thấp hơn kỳ vọng";
   399→  } else {
   400→    level = "warn";
   401→    hint = "cao — cần kiểm tra merge-split";
   402→  }
   403→  out.push({
   404→    key: "divRate",
   405→    label: "phân bào/100 khung",
   406→    level,
   407→    value: d.toFixed(1),
   408→    hint,
   409→  });
   410→
   411→  return out;
   412→}

   414→function analyzeCsv(text: string, fileName: string | null): AnalysisOutput {
   415→  const started = performance.now();
   416→  const lines = text.split(/\r\n|\r|\n/);
   417→
   418→  const firstLine = (lines[0] ?? "").replace(/^\uFEFF/, "");
   419→  const firstFields = parseCsvLine(firstLine);
   420→  const hasHeader = firstFields.some((f) => f.trim().toLowerCase() === "row_type");
   421→
   422→  const col: Record<string, number> = {};
   423→  if (hasHeader) {
   424→    firstFields.forEach((f, i) => {
   425→      const name = f.trim().toLowerCase();
   426→      if (!(name in col)) col[name] = i;
   427→    });
   428→    const missing = REQUIRED_COLS.filter((c) => !(c in col));
   429→    if (missing.length > 0) {
   430→      return {
   431→        ok: false,
   432→        result: null,
   433→        fatalIssues: [
   434→          {
   435→            severity: "error",
   436→            title: "Thiếu cột bắt buộc",
   437→            detail: `CSV thiếu các cột: ${missing.join(", ")}. Chuẩn: id, dataset, row_type, node_id, t, z, y, x, source_id, target_id (cột id có thể vắng).`,
   438→            rows: [],
   439→          },
   440→        ],
   441→      };
   442→    }
   443→  } else {
   444→    const n = firstFields.length;
   445→    if (n === 10) {
   446→      REQUIRED_COLS.forEach((name, i) => {
   447→        col[name] = i + 1;
   448→      });
   449→    } else if (n === 9) {
   450→      REQUIRED_COLS.forEach((name, i) => {
   451→        col[name] = i;
   452→      });
   453→    } else {
   454→      return {
   455→        ok: false,
   456→        result: null,
   457→        fatalIssues: [
   458→          {
   459→            severity: "error",
   460→            title: "Không nhận dạng được định dạng",
   461→            detail: `Dòng đầu tiên có ${n} cột và không chứa header (không tìm thấy cột "row_type"). Chuẩn: 10 cột (có cột id) hoặc 9 cột (không id).`,
   462→            rows: [],
   463→          },
   464→        ],
   465→      };
   466→    }
   467→  }
   468→
   469→  const drafts = new Map<string, IssueDraft>();
   470→  const pushIssue = (
   471→    key: string,
   472→    severity: Severity,
   473→    title: string,
   474→    detail: string,
   475→    row: number
   476→  ): void => {
   477→    const existing = drafts.get(key);
   478→    if (existing) {
   479→      existing.rows.push(row);
   480→      return;
   481→    }
   482→    drafts.set(key, { severity, title, detail, rows: [row] });
   483→  };
   484→
   485→  const idx = (name: string): number => col[name] ?? -1;
   486→  const getField = (fields: string[], i: number): string =>
   487→    i >= 0 && i < fields.length ? fields[i] ?? "" : "";
   488→
   489→  const nodesByDataset = new Map<string, Map<number, RawNode>>();
   490→  const rawEdges: RawEdge[] = [];
   491→  let nodeRows = 0;
   492→  let edgeRows = 0;
   493→  let totalRows = 0;
   494→
   495→  for (let li = hasHeader ? 1 : 0; li < lines.length; li++) {
   496→    const raw = lines[li] ?? "";
   497→    if (raw.trim() === "") continue;
   498→    totalRows++;
   499→    const lineNo = li + 1;
   500→    const fields = parseCsvLine(raw);
   501→    const rowType = getField(fields, idx("row_type")).trim().toLowerCase();
   502→    const dataset = getField(fields, idx("dataset")).trim();
   503→
   504→    if (rowType !== "node" && rowType !== "edge") {
   505→      pushIssue(
   506→        "row_type",
   507→        "error",
   508→        "row_type lạ",
   509→        `row_type phải là "node" hoặc "edge" — gặp "${rowType === "" ? "(rỗng)" : rowType}". Hàng bị bỏ qua.`,
   510→        lineNo
   511→      );
   512→      continue;
   513→    }
   514→    if (dataset === "") {
   515→      pushIssue(
   516→        "unparsable",
   517→        "error",
   518→        "Thiếu tên dataset",
   519→        "Cột dataset trống — không thể gán node/cạnh vào dataset nào. Hàng bị bỏ qua.",
   520→        lineNo
   521→      );
   522→      continue;
   523→    }
   524→
   525→    if (rowType === "node") {
   526→      nodeRows++;
   527→      const nodeIdRaw = getField(fields, idx("node_id")).trim();
   528→      const nodeId = toNum(nodeIdRaw);
   529→      if (nodeId === null || !Number.isInteger(nodeId) || nodeId <= 0) {
   530→        pushIssue(
   531→          "node_id",
   532→          "error",
   533→          "node_id không hợp lệ (≤ 0 hoặc không nguyên)",
   534→          "node_id phải là số nguyên dương. Hàng bị bỏ qua.",
   535→          lineNo
   536→        );
   537→        continue;
   538→      }
   539→      const t = toNum(getField(fields, idx("t")));
   540→      const z = toNum(getField(fields, idx("z")));
   541→      const y = toNum(getField(fields, idx("y")));
   542→      const x = toNum(getField(fields, idx("x")));
   543→      if (t === null || z === null || y === null || x === null) {
   544→        pushIssue(
   545→          "unparsable",
   546→          "error",
   547→          "Node thiếu tọa độ t/z/y/x",
   548→          "Các cột t, z, y, x của node phải là số. Hàng bị bỏ qua.",
   549→          lineNo
   550→        );
   551→        continue;
   552→      }
   553→      if (
   554→        t < 0 ||
   555→        z < 0 ||
   556→        y < 0 ||
   557→        x < 0 ||
   558→        !Number.isInteger(t) ||
   559→        !Number.isInteger(z) ||
   560→        !Number.isInteger(y) ||
   561→        !Number.isInteger(x)
   562→      ) {
   563→        pushIssue(
   564→          "coords",
   565→          "warning",
   566→          "Node có t/z/y/x âm hoặc không nguyên",
   567→          "Định dạng chuẩn của cuộc thi dùng chỉ số nguyên không âm (voxel). Node vẫn được tính vào thống kê.",
   568→          lineNo
   569→        );
   570→      }
   571→      let dsNodes = nodesByDataset.get(dataset);
   572→      if (!dsNodes) {
   573→        dsNodes = new Map();
   574→        nodesByDataset.set(dataset, dsNodes);
   575→      }
   576→      if (dsNodes.has(nodeId)) {
   577→        pushIssue(
   578→          "dup_node",
   579→          "error",
   580→          "node_id trùng trong 1 dataset",
   581→          `node_id ${nodeIdRaw} bị khai báo lần 2 trong cùng dataset — chỉ giữ node đầu tiên.`,
   582→          lineNo
   583→        );
   584→        continue;
   585→      }
   586→      dsNodes.set(nodeId, { dataset, nodeId, t, z, y, x, line: lineNo });
   587→    } else {
   588→      edgeRows++;
   589→      const source = toNum(getField(fields, idx("source_id")));
   590→      const target = toNum(getField(fields, idx("target_id")));
   591→      if (
   592→        source === null ||
   593→        target === null ||
   594→        !Number.isInteger(source) ||
   595→        !Number.isInteger(target)
   596→      ) {
   597→        pushIssue(
   598→          "unparsable",
   599→          "error",
   600→          "Cạnh thiếu source_id/target_id hợp lệ",
   601→          "source_id và target_id phải là số nguyên. Hàng bị bỏ qua.",
   602→          lineNo
   603→        );
   604→        continue;
   605→      }
   606→      rawEdges.push({ dataset, source, target, line: lineNo });
   607→    }
   608→  }
   609→
   610→  if (totalRows === 0) {
   611→    return {
   612→      ok: false,
   613→      result: null,
   614→      fatalIssues: [
   615→        {
   616→          severity: "error",
   617→          title: "Không có hàng dữ liệu",
   618→          detail: "Sau khi bỏ header và dòng trống, không còn hàng dữ liệu nào để phân tích.",
   619→          rows: [],
   620→        },
   621→      ],
   622→    };
   623→  }
   624→
   625→  const findOtherDataset = (nodeId: number, exclude: string): string | null => {
   626→    for (const [name, m] of nodesByDataset) {
   627→      if (name !== exclude && m.has(nodeId)) return name;
   628→    }
   629→    return null;
   630→  };
   631→
   632→  const validEdges: ValidEdge[] = [];
   633→  const edgeSeen = new Set<string>();
   634→
   635→  for (const e of rawEdges) {
   636→    if (e.source === e.target) {
   637→      pushIssue(
   638→        "self_loop",
   639→        "error",
   640→        "Cạnh tự nối (source = target)",
   641→        "Cạnh nối node với chính nó — vô nghĩa trong tracking. Cạnh bị loại.",
   642→        e.line
   643→      );
   644→      continue;
   645→    }
   646→    const dsNodes = nodesByDataset.get(e.dataset);
   647→    const srcOk = dsNodes !== undefined && dsNodes.has(e.source);
   648→    const tgtOk = dsNodes !== undefined && dsNodes.has(e.target);
   649→    if (!srcOk || !tgtOk) {
   650→      const missing: string[] = [];
   651→      if (!srcOk) {
   652→        const other = findOtherDataset(e.source, e.dataset);
   653→        if (other) {
   654→          pushIssue(
   655→            "cross_ds",
   656→            "error",
   657→            "Cạnh nối node khác dataset",
   658→            `Cạnh khai báo dataset ${e.dataset} nhưng node ${e.source} thuộc dataset ${other} — cạnh bị loại.`,
   659→            e.line
   660→          );
   661→        } else {
   662→          missing.push(`source_id ${e.source}`);
   663→        }
   664→      }
   665→      if (!tgtOk) {
   666→        const other = findOtherDataset(e.target, e.dataset);
   667→        if (other) {
   668→          pushIssue(
   669→            "cross_ds",
   670→            "error",
   671→            "Cạnh nối node khác dataset",
   672→            `Cạnh khai báo dataset ${e.dataset} nhưng node ${e.target} thuộc dataset ${other} — cạnh bị loại.`,
   673→            e.line
   674→          );
   675→        } else {
   676→          missing.push(`target_id ${e.target}`);
   677→        }
   678→      }
   679→      if (missing.length > 0) {
   680→        pushIssue(
   681→          "dangling",
   682→          "error",
   683→          "Cạnh treo lơ lửng (node không tồn tại)",
   684→          `Không tìm thấy node: ${missing.join(", ")} trong dataset ${e.dataset} — cạnh bị loại.`,
   685→          e.line
   686→        );
   687→      }
   688→      continue;
   689→    }
   690→    const key = `${e.dataset}|${e.source}|${e.target}`;
   691→    if (edgeSeen.has(key)) {
   692→      pushIssue(
   693→        "dup_edge",
   694→        "warning",
   695→        "Cạnh trùng lặp",
   696→        "Cặp source-target xuất hiện hơn một lần — chỉ tính 1 lần.",
   697→        e.line
   698→      );
   699→      continue;
   700→    }
   701→    edgeSeen.add(key);
   702→    validEdges.push(e);
   703→  }
   704→
   705→  const datasetNames = [...nodesByDataset.keys()].sort();
   706→  const datasets: DatasetAnalysis[] = [];
   707→  const allTrackLengths: number[] = [];
   708→  let globalNodes = 0;
   709→  let globalEdges = 0;
   710→  let globalDivisions = 0;
   711→  let globalSpan = 0;
   712→  let globalNoIn = 0;
   713→
   714→  for (const name of datasetNames) {
   715→    const dsNodes = nodesByDataset.get(name) ?? new Map<number, RawNode>();
   716→    const nodes = [...dsNodes.values()];
   717→    const edges = validEdges.filter((e) => e.dataset === name);
   718→
   719→    const outLines = new Map<number, number[]>();
   720→    const inLines = new Map<number, number[]>();
   721→    for (const e of edges) {
   722→      const out = outLines.get(e.source);
   723→      if (out) out.push(e.line);
   724→      else outLines.set(e.source, [e.line]);
   725→      const inn = inLines.get(e.target);
   726→      if (inn) inn.push(e.line);
   727→      else inLines.set(e.target, [e.line]);
   728→
   729→      const srcNode = dsNodes.get(e.source);
   730→      const tgtNode = dsNodes.get(e.target);
   731→      if (srcNode && tgtNode && tgtNode.t !== srcNode.t + 1) {
   732→        pushIssue(
   733→          "time_jump",
   734→          "warning",
   735→          "Cạnh nhảy thời gian (t đích ≠ t nguồn + 1)",
   736→          "Metric chính thức BỎ hẳn cạnh t→t+k khi chấm — cạnh nhảy không phải FP nhưng cũng KHÔNG BAO GIỜ là TP; phải nội suy node giữa.",
   737→          e.line
   738→        );
   739→      }
   740→    }
   741→    for (const linesArr of outLines.values()) {
   742→      if (linesArr.length > 2) {
   743→        for (const ln of linesArr) {
   744→          pushIssue(
   745→            "out_deg",
   746→            "warning",
   747→            "Node có cạnh ra > 2",
   748→            "Metric chỉ coi tối đa 2 cạnh ra mỗi node (fork = phân bào); phần thừa bị bỏ.",
   749→            ln
   750→          );
   751→        }
   752→      }
   753→    }
   754→    for (const linesArr of inLines.values()) {
   755→      if (linesArr.length > 1) {
   756→        for (const ln of linesArr) {
   757→          pushIssue(
   758→            "in_deg",
   759→            "warning",
   760→            "Node có cạnh vào > 1",
   761→            "Hợp nhất 2 tế bào = 1 track, sinh học vô lý.",
   762→            ln
   763→          );
   764→        }
   765→      }
   766→    }
   767→
   768→    // thành phần liên thông yếu (union-find)
   769→    const parent = new Map<number, number>();
   770→    for (const n of nodes) parent.set(n.nodeId, n.nodeId);
   771→    const find = (x: number): number => {
   772→      let root = x;
   773→      for (;;) {
   774→        const p = parent.get(root) ?? root;
   775→        if (p === root) break;
   776→        root = p;
   777→      }
   778→      let cur = x;
   779→      while (cur !== root) {
   780→        const p = parent.get(cur) ?? root;
   781→        parent.set(cur, root);
   782→        cur = p;
   783→      }
   784→      return root;
   785→    };
   786→    for (const e of edges) {
   787→      parent.set(find(e.source), find(e.target));
   788→    }
   789→    const sizeByRoot = new Map<number, number>();
   790→    for (const n of nodes) {
   791→      const r = find(n.nodeId);
   792→      sizeByRoot.set(r, (sizeByRoot.get(r) ?? 0) + 1);
   793→    }
   794→    const trackLengths = [...sizeByRoot.values()];
   795→
   796→    const nodeCount = nodes.length;
   797→    const edgeCount = edges.length;
   798→    let divisions = 0;
   799→    for (const linesArr of outLines.values()) {
   800→      if (linesArr.length >= 2) divisions++;
   801→    }
   802→    let noIn = 0;
   803→    for (const n of nodes) {
   804→      if ((inLines.get(n.nodeId) ?? []).length === 0) noIn++;
   805→    }
   806→    let minT = Number.POSITIVE_INFINITY;
   807→    let maxT = Number.NEGATIVE_INFINITY;
   808→    for (const n of nodes) {
   809→      if (n.t < minT) minT = n.t;
   810→      if (n.t > maxT) maxT = n.t;
   811→    }
   812→    const span =
   813→      nodes.length > 0 && Number.isFinite(minT) && Number.isFinite(maxT)
   814→        ? maxT - minT + 1
   815→        : 0;
   816→    const singleTracks = trackLengths.filter((l) => l === 1).length;
   817→    const trackCount = trackLengths.length;
   818→    const nodesPerFrame = span > 0 ? nodeCount / span : nodeCount;
   819→    const edgesPerNode = nodeCount > 0 ? edgeCount / nodeCount : 0;
   820→    const pctNoInEdge = nodeCount > 0 ? (noIn / nodeCount) * 100 : 0;
   821→    const pctSingleTrack = trackCount > 0 ? (singleTracks / trackCount) * 100 : 0;
   822→    const medianTrackLen = median(trackLengths);
   823→    const divisionRate = span > 0 ? (divisions / span) * 100 : 0;
   824→
   825→    datasets.push({
   826→      name,
   827→      nodes: nodeCount,
   828→      edges: edgeCount,
   829→      divisions,
   830→      frameLabel:
   831→        nodes.length > 0 && span > 0 ? `${minT}–${maxT}` : "—",
   832→      frameSpan: span,
   833→      nodesPerFrame,
   834→      edgesPerNode,
   835→      pctNoInEdge,
   836→      pctSingleTrack,
   837→      trackCount,
   838→      medianTrackLen,
   839→      divisionRate,
   840→      trackLengths,
   841→      badges: assessBadges({
   842→        edgesPerNode,
   843→        pctNoInEdge,
   844→        pctSingleTrack,
   845→        medianTrackLen,
   846→        divisionRate,
   847→        nodesPerFrame,
   848→      }),
   849→    });
   850→
   851→    globalNodes += nodeCount;
   852→    globalEdges += edgeCount;
   853→    globalDivisions += divisions;
   854→    globalSpan += span;
   855→    globalNoIn += noIn;
   856→    allTrackLengths.push(...trackLengths);
   857→  }
   858→
   859→  const global: DatasetAnalysis = {
   860→    name: "Σ toàn cục",
   861→    nodes: globalNodes,
   862→    edges: globalEdges,
   863→    divisions: globalDivisions,
   864→    frameLabel: `Σ ${globalSpan}`,
   865→    frameSpan: globalSpan,
   866→    nodesPerFrame: globalSpan > 0 ? globalNodes / globalSpan : globalNodes,
   867→    edgesPerNode: globalNodes > 0 ? globalEdges / globalNodes : 0,
   868→    pctNoInEdge: globalNodes > 0 ? (globalNoIn / globalNodes) * 100 : 0,
   869→    pctSingleTrack:
   870→      allTrackLengths.length > 0
   871→        ? (allTrackLengths.filter((l) => l === 1).length / allTrackLengths.length) * 100
   872→        : 0,
   873→    trackCount: allTrackLengths.length,
   874→    medianTrackLen: median(allTrackLengths),
   875→    divisionRate: globalSpan > 0 ? (globalDivisions / globalSpan) * 100 : 0,
   876→    trackLengths: allTrackLengths,
   877→    badges: [],
   878→  };
   879→  global.badges = assessBadges({
   880→    edgesPerNode: global.edgesPerNode,
   881→    pctNoInEdge: global.pctNoInEdge,
   882→    pctSingleTrack: global.pctSingleTrack,
   883→    medianTrackLen: global.medianTrackLen,
   884→    divisionRate: global.divisionRate,
   885→    nodesPerFrame: global.nodesPerFrame,
   886→  });
   887→
   888→  const issues: Issue[] = [...drafts.values()].map((d) => ({
   889→    severity: d.severity,
   890→    title: d.title,
   891→    detail: d.detail,
   892→    rows: d.rows,
   893→  }));
   894→  const order: Record<Severity, number> = { error: 0, warning: 1 };
   895→  issues.sort((a, b) => order[a.severity] - order[b.severity]);
   896→
   897→  return {
   898→    ok: true,
   899→    result: {
   900→      datasets,
   901→      global,
   902→      issues,
   903→      errorCount: issues.filter((i) => i.severity === "error").length,
   904→      warningCount: issues.filter((i) => i.severity === "warning").length,
   905→      totalRows,
   906→      nodeRows,
   907→      edgeRows,
   908→      parseMs: Math.max(1, Math.round(performance.now() - started)),
   909→      hasHeader,
   910→      fileName,
   911→    },
   912→    fatalIssues: [],
   913→  };
   914→}

   920→interface ModelResult {
   921→  M: number;
   922→  Npred: number;
   923→  S: number;
   924→  TP: number;
   925→  FN: number;
   926→  FP: number;
   927→  EJ: number;
   928→  adjEJ: number;
   929→  score: number;
   930→}
   931→
   932→function computeModel(r: number, p: number, lambda: number, divJ: number): ModelResult {
   933→  const N = 1000;
   934→  const q = 0.92;
   935→  const M = (r / 100) * N;
   936→  const Npred = M / (p / 100);
   937→  const S = Npred - M;
   938→  const Egt = q * N;
   939→  const TP = (r / 100) * (r / 100) * (lambda / 100) * Egt;
   940→  const FN = Egt - TP;
   941→  const FP = q * Npred - TP;
   942→  const denom = TP + FP + FN;
   943→  const EJ = denom > 0 ? TP / denom : 0;
   944→  const adjEJ = EJ * Math.max(0, 1 - (0.1 * (Npred - N)) / N);
   945→  const score = adjEJ + 0.1 * (divJ / 100);
   946→  return { M, Npred, S, TP, FN, FP, EJ, adjEJ, score };
   947→}
