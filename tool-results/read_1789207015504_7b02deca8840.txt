   133→function genDetections(sim: SimData, p: Params, seed: number): Det[][] {
   134→  const rng = mulberry32(seed ^ 0x51ab3f)
   135→  const out: Det[][] = []
   136→  for (let t = 0; t < T; t++) {
   137→    const dets: Det[] = []
   138→    for (const tr of sim.tracks) {
   139→      if (tr.start > t || tr.end < t) continue
   140→      if (rng() > p.recall) continue
   141→      const f = tr.frames[t - tr.start]
   142→      dets.push({
   143→        x: clamp(f.x + gauss(rng) * p.jitter, 3, WORLD_W - 3),
   144→        y: clamp(f.y + gauss(rng) * p.jitter, 3, WORLD_H - 3),
   145→        z: clamp(f.z + gauss(rng) * p.jitter * 0.5, 1, 62),
   146→        r: f.r,
   147→        i: f.i,
   148→        mass: Math.round(f.i * 27),
   149→        src: tr.id,
   150→      })
   151→    }
   152→    // node giả (false positive) rải ngẫu nhiên
   153→    const nSpur = Math.round(dets.length * p.fpRate * (0.6 + rng() * 0.8))
   154→    for (let k = 0; k < nSpur; k++) {
   155→      dets.push({
   156→        x: 8 + rng() * (WORLD_W - 16),
   157→        y: 6 + rng() * (WORLD_H - 12),
   158→        z: 2 + rng() * 58,
   159→        r: 1.6 + rng() * 1.6,
   160→        i: 500 + Math.round(rng() * 1500),
   161→        mass: 18000,
   162→        src: null,
   163→      })
   164→    }
   165→    out.push(dets)
   166→  }
   167→  return out
   168→}

   170→/** Chấm điểm chế độ tùy chỉnh: greedy nearest-neighbor + phân bào giả lập */
   171→function analyzeCustom(
   172→  sim: SimData,
   173→  gt: GtGraph,
   174→  p: Params,
   175→  seed: number,
   176→): Analysis {
   177→  const det = genDetections(sim, p, seed)
   178→  const edgeRefs: EdgeRef[] = []
   179→  for (let t = 0; t < T - 1; t++) {
   180→    const A = det[t]!
   181→    const B = det[t + 1]!
   182→    const pairs: { ai: number; bi: number; d: number }[] = []
   183→    for (let ai = 0; ai < A.length; ai++) {
   184→      for (let bi = 0; bi < B.length; bi++) {
   185→        const da = A[ai]!
   186→        const db = B[bi]!
   187→        const d = dist3(da.x, da.y, da.z, db.x, db.y, db.z)
   188→        if (d <= p.linkUm) pairs.push({ ai, bi, d })
   189→      }
   190→    }
   191→    pairs.sort((x, y) => x.d - y.d)
   192→    const usedA = new Set<number>()
   193→    const usedB = new Set<number>()
   194→    const primary = new Map<number, number>()
   195→    for (const pr of pairs) {
   196→      if (usedA.has(pr.ai) || usedB.has(pr.bi)) continue
   197→      usedA.add(pr.ai)
   198→      usedB.add(pr.bi)
   199→      primary.set(pr.ai, pr.bi)
   200→      edgeRefs.push({ at: t, ai: pr.ai, bt: t + 1, bi: pr.bi })
   201→    }
   202→    // phát hiện phân bào: cho phép 1 node nối thêm "con thứ 2" nằm gần
   203→    if (p.division) {
   204→      for (const [ai, bi1] of primary) {
   205→        const da = A[ai]!
   206→        const db1 = B[bi1]!
   207→        for (let bi = 0; bi < B.length; bi++) {
   208→          if (usedB.has(bi)) continue
   209→          const db = B[bi]!
   210→          const d2 = dist3(da.x, da.y, da.z, db.x, db.y, db.z)
   211→          const d3v = dist3(db1.x, db1.y, db1.z, db.x, db.y, db.z)
   212→          if (d2 <= p.linkUm && d3v <= p.linkUm) {
   213→            usedB.add(bi)
   214→            edgeRefs.push({ at: t, ai, bt: t + 1, bi })
   215→          }
   216→        }
   217→      }
   218→    }
   219→  }
   220→  return scoreDetections(sim, gt, det, edgeRefs)
   221→}

   230→function createNoiseCanvas(): HTMLCanvasElement {
   231→  const nc = document.createElement('canvas')
   232→  nc.width = 128
   233→  nc.height = 128
   234→  const nctx = nc.getContext('2d')
   235→  if (nctx !== null) {
   236→    const img = nctx.createImageData(128, 128)
   237→    const nrng = mulberry32(0x1234abcd)
   238→    for (let k = 0; k < img.data.length; k += 4) {
   239→      const val = 8 + Math.floor(nrng() * 14)
   240→      img.data[k] = val
   241→      img.data[k + 1] = val + 4
   242→      img.data[k + 2] = val
   243→      img.data[k + 3] = 255
   244→    }
   245→    nctx.putImageData(img, 0, 0)
   246→  }
   247→  return nc
   248→}
