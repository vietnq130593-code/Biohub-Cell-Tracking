def sha(path):
    digest = hashlib.sha256()
    with path.open('rb') as stream:
        for chunk in iter(lambda: stream.read(1024**2), b''):
            digest.update(chunk)
    return digest.hexdigest()
runtime = runtimes[0]

    raise ValueError('Attached expanded dataset contract changed')
