   167→function genDetections(sim: SimData, p: Params, seed: number): Det[][] {
   168→  const rng = mulberry32(seed ^ 0x51ab3f)
   169→  const out: Det[][] = []
   170→  for (let t = 0; t < T; t++) {
   171→    const dets: Det[] = []
   172→    for (const tr of sim.tracks) {
   173→      if (tr.start > t || tr.end < t) continue
   174→      if (rng() > p.recall) continue
   175→      const f = tr.frames[t - tr.start]
   176→      dets.push({
   177→        x: clamp(f.x + gauss(rng) * p.jitter, 3, WORLD_W - 3),
   178→        y: clamp(f.y + gauss(rng) * p.jitter, 3, WORLD_H - 3),
   179→        z: clamp(f.z + gauss(rng) * p.jitter * 0.5, 1, 62),
   180→        r: f.r,
   181→        i: f.i,
   182→        mass: Math.round(f.i * 27),
   183→        src: tr.id,
   184→      })
   185→    }
   186→    // node giả (false positive) rải ngẫu nhiên
   187→    const nSpur = Math.round(dets.length * p.fpRate * (0.6 + rng() * 0.8))
   188→    for (let k = 0; k < nSpur; k++) {
   189→      dets.push({
   190→        x: 8 + rng() * (WORLD_W - 16),
   191→        y: 6 + rng() * (WORLD_H - 12),
   192→        z: 2 + rng() * 58,
   193→        r: 1.6 + rng() * 1.6,
   194→        i: 500 + Math.round(rng() * 1500),
   195→        mass: 18000,
   196→        src: null,
   197→      })
   198→    }
   199→    out.push(dets)
   200→  }
   201→  return out
   202→}

   204→/** Chấm điểm chế độ tùy chỉnh: greedy nearest-neighbor + phân bào giả lập */
   205→function analyzeCustom(
   206→  sim: SimData,
   207→  gt: GtGraph,
   208→  p: Params,
   209→  seed: number,
   210→): Analysis {
   211→  const det = genDetections(sim, p, seed)
   212→  const edgeRefs: EdgeRef[] = []
   213→  for (let t = 0; t < T - 1; t++) {
   214→    const A = det[t]!
   215→    const B = det[t + 1]!
   216→    const pairs: { ai: number; bi: number; d: number }[] = []
   217→    for (let ai = 0; ai < A.length; ai++) {
   218→      for (let bi = 0; bi < B.length; bi++) {
   219→        const da = A[ai]!
   220→        const db = B[bi]!
   221→        const d = dist3(da.x, da.y, da.z, db.x, db.y, db.z)
   222→        if (d <= p.linkUm) pairs.push({ ai, bi, d })
   223→      }
   224→    }
   225→    pairs.sort((x, y) => x.d - y.d)
   226→    const usedA = new Set<number>()
   227→    const usedB = new Set<number>()
   228→    const primary = new Map<number, number>()
   229→    for (const pr of pairs) {
   230→      if (usedA.has(pr.ai) || usedB.has(pr.bi)) continue
   231→      usedA.add(pr.ai)
   232→      usedB.add(pr.bi)
   233→      primary.set(pr.ai, pr.bi)
   234→      edgeRefs.push({ at: t, ai: pr.ai, bt: t + 1, bi: pr.bi })
   235→    }
   236→    // phát hiện phân bào: cho phép 1 node nối thêm "con thứ 2" nằm gần
   237→    if (p.division) {
   238→      for (const [ai, bi1] of primary) {
   239→        const da = A[ai]!
   240→        const db1 = B[bi1]!
   241→        for (let bi = 0; bi < B.length; bi++) {
   242→          if (usedB.has(bi)) continue
   243→          const db = B[bi]!
   244→          const d2 = dist3(da.x, da.y, da.z, db.x, db.y, db.z)
   245→          const d3v = dist3(db1.x, db1.y, db1.z, db.x, db.y, db.z)
   246→          if (d2 <= p.linkUm && d3v <= p.linkUm) {
   247→            usedB.add(bi)
   248→            edgeRefs.push({ at: t, ai, bt: t + 1, bi })
   249→          }
   250→        }
   251→      }
   252→    }
   253→  }
   254→  return scoreDetections(sim, gt, det, edgeRefs)
   255→}

   264→function createNoiseCanvas(): HTMLCanvasElement {
   265→  const nc = document.createElement('canvas')
   266→  nc.width = 128
   267→  nc.height = 128
   268→  const nctx = nc.getContext('2d')
   269→  if (nctx !== null) {
   270→    const img = nctx.createImageData(128, 128)
   271→    const nrng = mulberry32(0x1234abcd)
   272→    for (let k = 0; k < img.data.length; k += 4) {
   273→      const val = 8 + Math.floor(nrng() * 14)
   274→      img.data[k] = val
   275→      img.data[k + 1] = val + 4
   276→      img.data[k + 2] = val
   277→      img.data[k + 3] = 255
   278→    }
   279→    nctx.putImageData(img, 0, 0)
   280→  }
   281→  return nc
   282→}
